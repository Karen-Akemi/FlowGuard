1) COLUNAS CALCULADAS NOVAS (Modelagem > Nova Coluna, na tabela 'Dataset Geral')
-----------------------------------------------------------------------

Periodo Comparacao =
VAR d = 'Dataset Geral'[Aberto].[Date]
RETURN
    SWITCH(
        TRUE(),
        d >= DATE(2025,12,2) && d <= DATE(2025,12,31), "Atual (30d)",
        d >= DATE(2025,11,2) && d <= DATE(2025,12,1),  "Anterior (30d)",
        "Fora do período"
    )

Faixa Duracao =
SWITCH(
    TRUE(),
    'Dataset Geral'[Duração] <= 86400,                     "≤ 24h",
    'Dataset Geral'[Duração] > 86400 && 'Dataset Geral'[Duração] <= 172800, "24–48h",
    'Dataset Geral'[Duração] > 172800,                     "> 48h"
)
-- Duração está em SEGUNDOS. 86400s = 24h, 172800s = 48h.

2) MEDIDAS NOVAS (Modelagem > Nova Medida, na tabela 'Medidas' ou 'Dataset Geral')
-----------------------------------------------------------------------

-- Total de ocorrências no período atual/anterior (30 dias) --

Total 30d Atual =
CALCULATE(
    [Total de Incidentes],
    'Dataset Geral'[Aberto] >= DATE(2025,12,2),
    'Dataset Geral'[Aberto] < DATE(2026,1,1)
)

Total 30d Anterior =
CALCULATE(
    [Total de Incidentes],
    'Dataset Geral'[Aberto] >= DATE(2025,11,2),
    'Dataset Geral'[Aberto] < DATE(2025,12,2)
)

Var % Total 30d =
DIVIDE([Total 30d Atual] - [Total 30d Anterior], [Total 30d Anterior])
-- Esperado: ~26.323 vs ~21.948 => +19,9%

-- Prioridade Alta --

Prioridade Alta 30d Atual =
CALCULATE(
    [Total de Incidentes],
    'Dataset Geral'[Prioridade] = "2 - Alta",
    'Dataset Geral'[Aberto] >= DATE(2025,12,2),
    'Dataset Geral'[Aberto] < DATE(2026,1,1)
)

Prioridade Alta 30d Anterior =
CALCULATE(
    [Total de Incidentes],
    'Dataset Geral'[Prioridade] = "2 - Alta",
    'Dataset Geral'[Aberto] >= DATE(2025,11,2),
    'Dataset Geral'[Aberto] < DATE(2025,12,2)
)

Var % Prioridade Alta =
DIVIDE([Prioridade Alta 30d Atual] - [Prioridade Alta 30d Anterior], [Prioridade Alta 30d Anterior])
-- Esperado: 1.806 vs 3.209 => -43,7%

-- Ocorrências acompanhadas (entraram para KPI) --

Acompanhadas 30d Atual =
CALCULATE(
    [Incidentes Elegíveis KPI],
    'Dataset Geral'[Aberto] >= DATE(2025,12,2),
    'Dataset Geral'[Aberto] < DATE(2026,1,1)
)
-- Se [Incidentes Elegíveis KPI] não filtrar por "Entrou para KPI?"="SIM", use:
-- CALCULATE([Total de Incidentes], 'Dataset Geral'[Entrou para KPI?]="SIM", <mesmas datas>)

Acompanhadas 30d Anterior =
CALCULATE(
    [Incidentes Elegíveis KPI],
    'Dataset Geral'[Aberto] >= DATE(2025,11,2),
    'Dataset Geral'[Aberto] < DATE(2025,12,2)
)
-- Esperado: 1.359 vs 1.662 => -18,2%

-- Violação de SLA --

KPIs Violados 30d Atual =
CALCULATE(
    [KPIs Violados],
    'Dataset Geral'[Aberto] >= DATE(2025,12,2),
    'Dataset Geral'[Aberto] < DATE(2026,1,1)
)

Taxa Violacao 30d Atual =
DIVIDE([KPIs Violados 30d Atual], [Acompanhadas 30d Atual])

KPIs Violados 30d Anterior =
CALCULATE(
    [KPIs Violados],
    'Dataset Geral'[Aberto] >= DATE(2025,11,2),
    'Dataset Geral'[Aberto] < DATE(2025,12,2)
)

Taxa Violacao 30d Anterior =
DIVIDE([KPIs Violados 30d Anterior], [Acompanhadas 30d Anterior])

Nao Violacao % Atual = 1 - [Taxa Violacao 30d Atual]
-- Esperado: ~98,68%

Tendencia Violacao =
VAR atual = [Taxa Violacao 30d Atual]
VAR anterior = [Taxa Violacao 30d Anterior]
RETURN
    SWITCH(
        TRUE(),
        ISBLANK(anterior) || atual > anterior + 0.005, "▲",
        atual < anterior - 0.005, "▼",
        "▬"
    )
-- Use na tabela "Equipes por Risco de SLA", agrupada por 'Dataset Geral'[Grupo designado]

Nivel Risco Equipe =
VAR taxa = [Taxa Violacao 30d Atual]
RETURN
    SWITCH(
        TRUE(),
        ISBLANK(taxa), "Sem dados",
        taxa >= 0.20, "Crítico",
        taxa >= 0.10, "Alto",
        taxa >= 0.03, "Médio",
        "Baixo"
    )

-- Mediana de duração só das ocorrências acompanhadas --

Mediana Duracao Acompanhadas (min) =
CALCULATE(
    [Mediana Duracao Minutos],
    'Dataset Geral'[Entrou para KPI?] = "SIM"
)
-- Esperado (30d atual): ~68,6 min (1h09m)
